kubectl exec fluent-bit-b9mkl -c fluent-bit -- sh -c '
for pat in \
  "*_airflow3_airflow*.log" \
  "*_airflow3qa_airflow*.log"
do
  echo "===== $pat ====="

  find /var/log/containers -maxdepth 1 \
    -name "$pat" -print 2>/dev/null | head -5

  printf "Matched files: "
  find /var/log/containers -maxdepth 1 \
    -name "$pat" -print 2>/dev/null | wc -l
done
'
