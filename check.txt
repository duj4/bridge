for pod in fluent-bit-b9mkl fluent-bit-fxn8n fluent-bit-qfxfs; do
  echo "========== $pod =========="

  kubectl logs "$pod" -c fluent-bit --since=24h |
    awk '
      /created new multiline stream for/ {
        if ($0 ~ /kube\.Airflow3qa\./)
          qa++
        else if ($0 ~ /kube\.Airflow3\./)
          airflow++
      }
      END {
        printf "Airflow3 streams created:    %d\n", airflow
        printf "Airflow3qa streams created:  %d\n", qa
      }
    '
done
