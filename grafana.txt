jq '.dashboards | length' mimirtool.json

jq -r '
  .dashboards[] |
  "\(.title) | \(.metrics | length) metrics"
' mimirtool.json

jq '{
  dashboards: (.dashboards | length),
  unique_dashboards: ([.dashboards[].uid] | unique | length),
  unique_metrics: (.metricsUsed | length),
  dashboards_with_metrics: (
    [.dashboards[] | select((.metrics | length) > 0)] | length
  ),
  dashboards_with_parse_errors: (
    [.dashboards[] | select((.parse_errors | length) > 0)] | length
  )
}' mimirtool.json

jq -r '
  [.dashboards[].parse_errors[]? |
    if startswith("unsupported panel type:") then .
    elif startswith("query=") then "PromQL query parse error"
    else "Other error"
    end
  ]
  | group_by(.)
  | map({type: .[0], count: length})
  | sort_by(-.count)[]
  | "\(.count)\t\(.type)"
' bjv-grafana-eut-metrics-in-dashboards.json

jq -r '
  .dashboards[]
  | [.parse_errors[]? | select(startswith("query="))] as $errors
  | select(($errors | length) > 0)
  | "\(.title) | \($errors | length) query errors"
' bjv-grafana-eut-metrics-in-dashboards.json

jq -r '
  .dashboards[] as $d
  | ($d.parse_errors // [])[]
  | select(startswith("query="))
  | ([scan("windows_[A-Za-z0-9_:]+")] | unique)[] as $m
  | select((($d.metrics // []) | index($m)) == null)
  | "\($d.title)\t\($m)"
' bjv-grafana-eut-metrics-in-dashboards.json | sort -u

jq -r '
  .dashboards[]
  | select((.metrics // [] | length) == 0)
  | "\(.title) | errors: \((.parse_errors // []) | length)"
' bjv-grafana-eut-metrics-in-dashboards.json

jq '{
  alert_rules: length,
  query_targets: ([.[] | .data[]?] | length)
}' grafana-alert-rules.json

jq '
  def is_expression:
    .datasourceUid == "__expr__"
    or .datasourceUid == "-100"
    or .model.datasource.type == "__expr__";

  def is_prometheus:
    .model.datasource.type == "prometheus";

  [.[] | .data[]?] as $targets |
  {
    all_targets: ($targets | length),
    prometheus_targets: (
      [$targets[] | select(is_prometheus)] | length
    ),
    grafana_expression_targets: (
      [$targets[] | select(is_expression)] | length
    ),
    other_or_unknown: (
      [$targets[] |
        select((is_prometheus or is_expression) | not)
      ] | length
    )
  }
' bjv-grafana-eut-alerts.json

jq '
  to_entries[] as $rule
  | ($rule.value.data // [] | to_entries[])
  | select(
      .value.datasourceUid != "__expr__"
      and .value.datasourceUid != "-100"
      and .value.model.datasource.type != "__expr__"
      and .value.model.datasource.type != "prometheus"
    )
  | {
      alert_index: $rule.key,
      refId: .value.refId,
      datasource_type: .value.model.datasource.type,
      model_type: .value.model.type,
      has_expr: ((.value.model.expr // "") | length > 0)
    }
' bjv-grafana-eut-alerts.json

jq '
  ([
    .[] | .data[]?
    | select(.model.datasource.type == "prometheus")
    | .datasourceUid
    | select(type == "string" and length > 0)
  ] | unique) as $prom_uids

  | to_entries[] as $rule
  | ($rule.value.data // [])[] as $t

  | select(
      .datasourceUid != "__expr__"
      and .datasourceUid != "-100"
      and .model.datasource.type != "__expr__"
      and .model.datasource.type != "prometheus"
    )

  | ($t.datasourceUid // $t.model.datasource.uid // "") as $uid

  | {
      alert_index: $rule.key,
      refId: $t.refId,
      datasource_uid_present: ($uid != ""),
      known_prometheus_uid: (
        $uid != "" and ($prom_uids | index($uid)) != null
      ),
      has_expr: (($t.model.expr // "") | length > 0)
    }
' bjv-grafana-eut-alerts.json
