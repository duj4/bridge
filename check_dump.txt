for pod in fluent-bit-fxn8n fluent-bit-b9mkl fluent-bit-qfxfs; do
  echo "========== $pod =========="

  kubectl exec "$pod" -c fluent-bit -- sh -c '
    DIR=/var/log/fluent-bit-heap-profiles

    echo "POD_NAME=$POD_NAME"

    if [ ! -d "$DIR" ]; then
      echo "ERROR: Profile directory missing"
      exit 0
    fi

    echo "=== CURRENT POD HEAP FILES ==="

    find "$DIR" \
      -maxdepth 1 \
      -type f \
      -name "${POD_NAME}.*.heap" \
      -printf "%TY-%Tm-%Td %TH:%TM:%TS %s bytes %f\n" |
      sort

    echo "=== CURRENT POD HEAP COUNT ==="

    find "$DIR" \
      -maxdepth 1 \
      -type f \
      -name "${POD_NAME}.*.heap" |
      wc -l
  '
done
