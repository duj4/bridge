cp -p flb-jeprof-isolated.sh \
  "flb-jeprof-isolated.sh.bak.$(date +%Y%m%d%H%M%S)"

python3 - <<'PY'
from pathlib import Path

p = Path("flb-jeprof-isolated.sh")
s = p.read_text()

assert 'TRANSFER=("$FIRST")' in s
assert '# === FULL-STACK ATTRIBUTION V4 ===' in s
assert 'echo "=== ANALYSIS COMPLETE ==="' in s

# Transfer every completed profile belonging to the current process.
start = s.index('TRANSFER=("$FIRST")')
end = s.index('echo "First: $FIRST"', start)

s = (
    s[:start]
    + 'TRANSFER=("${FILES[@]}")\n\n'
    + s[end:]
)

# Replace verbose V4 full-stack output with compact V5 time series.
start = s.index('# === FULL-STACK ATTRIBUTION V4 ===')
end = s.index('echo "=== ANALYSIS COMPLETE ==="', start)

block = r'''
# === HEAP TIME SERIES V5 ===

echo '=== HEAP TIME SERIES V5 ==='
echo 'All values are sampled estimated in-use MiB.'
echo 'Categories are heuristic, mutually exclusive stack classifications.'
echo

printf 'PROFILE\tUTC\tTOTAL\tLOKI_TLS\tTAIL\tMETRICS\tCHUNK\tOTHER\n'

find /work/profiles -maxdepth 1 -type f \
  -name '*.i*.heap' \
  -printf '%f\n' |
  sort -V > /work/timeseries-files.txt

while IFS= read -r name; do

  profile="/work/profiles/$name"

  utc="$(date -u -d "@$(stat -c %Y "$profile")" \
    '+%Y-%m-%dT%H:%M:%SZ')"

  if ! timeout 120 jeprof \
    --collapsed \
    --inuse_space \
    --show_bytes \
    --exclude="$FILTER" \
    "$BIN" "$profile" \
    > /work/current.collapsed \
    2> /work/current.errors; then

    echo "ERROR: jeprof failed for $name"
    cat /work/current.errors
    exit 41
  fi

  awk -v name="$name" -v utc="$utc" '
    NF >= 2 && $NF ~ /^[0-9]+$/ {
      bytes = $NF + 0
      total += bytes

      if ($0 ~ /cb_loki_flush|send_loki_payload|flb_tls_session_create/) {
        loki += bytes
      }
      else if ($0 ~ /flb_tail_file_append|flb_tail_scan|flb_tail_file_chunk|in_tail_collect|tail_fs_event/) {
        tail += bytes
      }
      else if ($0 ~ /cb_metrics_prometheus|unpack_metric_array_entry|cmt_|flb_hs_push/) {
        metrics += bytes
      }
      else if ($0 ~ /cio_memfs_open|flb_input_chunk_create/) {
        chunk += bytes
      }
      else {
        other += bytes
      }
    }

    END {
      unit = 1048576
      printf "%s\t%s\t%.2f\t%.2f\t%.2f\t%.2f\t%.2f\t%.2f\n",
        name, utc, total/unit, loki/unit, tail/unit,
        metrics/unit, chunk/unit, other/unit
    }
  ' /work/current.collapsed

done < /work/timeseries-files.txt

echo '=== END HEAP TIME SERIES ==='

'''

s = s[:start] + block + s[end:]

p.write_text(s)
print("V5 time-series analysis enabled.")
PY

bash -n flb-jeprof-isolated.sh &&
  echo "BASH SYNTAX PASSED"
