import csv
from pathlib import Path

root = Path.home() / "flb-observations"

old = root / "20261029T075727Z"
new = root / "20261031T014544Z"

keys = [
    "cgroup_memory.current",
    "cgroup_memory.peak",
    "cgroup_anon_bytes",
    "cgroup_file_bytes",
    "cgroup_kernel_bytes",
    "cgroup_slab_bytes",
    "cgroup_slab_reclaimable_bytes",
    "cgroup_slab_unreclaimable_bytes",
    "cgroup_pagetables_bytes",
    "cgroup_sock_bytes",
    "cgroup_inactive_file_bytes",
    "cgroup_active_file_bytes",
    "events_max",
    "events_oom",
    "events_oom_kill",
]

def load(path):
    with path.open() as f:
        return {
            r["metric"]: r["value"]
            for r in csv.DictReader(f, delimiter="\t")
        }

for pod in (
    "fluent-bit-b9mkl",
    "fluent-bit-fxn8n",
    "fluent-bit-qfxfs",
):
    a = load(old / pod / "memory.tsv")
    b = load(new / pod / "memory.tsv")

    print(f"\n========== {pod} ==========")

    for key in keys:
        if key not in a or key not in b:
            continue
        if a[key] == "max" or b[key] == "max":
            continue

        x, y = float(a[key]), float(b[key])

        if key.startswith("events_"):
            print(f"{key:39s} {int(x):6d} -> {int(y):6d} ({y-x:+.0f})")
        else:
            unit = 1048576
            print(
                f"{key:39s} "
                f"{x/unit:8.2f} -> {y/unit:8.2f} MiB "
                f"({(y-x)/unit:+.2f} MiB)"
            )
