jq '
  .nodes
  | group_by(.type)
  | map({
      type: .[0].type,
      fields: (
        [.[] | to_entries[]
         | {field: .key, kind: (.value | type)}]
        | unique
      ),
      properties: (
        [.[] | .properties | objects | to_entries[]
         | {field: .key, kind: (.value | type)}]
        | unique
      )
    })
' graph/node.json
