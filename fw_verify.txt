jq '{
  topLevelKeys: keys_unsorted,
  topLevelTypes: with_entries(.value |= type),
  arrayPaths: [
    paths(type == "array")
    | map(tostring)
    | join(".")
  ]
}' filter/relationship_audit.json

for f in filter/related_objects.json filter/historical_flows.json; do
    echo "========== $f =========="

    jq '{
      topLevelKeys: keys_unsorted,
      dataKeys: (
        if (.data | type) == "object"
        then (.data | keys_unsorted)
        else []
        end
      ),
      arrayPaths: [
        paths(type == "array")
        | map(tostring)
        | join(".")
      ]
    }' "$f"
done

jq '.' filter/relationship_audit.json
