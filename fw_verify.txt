jq -n \
  --slurpfile raw objects.json \
  --slurpfile p filter/objects.json \
  --slurpfile rel filter/related_objects.json '
  def tokens($v):
    (($v // "") | tostring
    | split(",")
    | map(gsub("^\\s+|\\s+$"; ""))
    | map(select(length > 0)));

  ($raw[0].data.requested_objects.list) as $all

  | (
      $p[0].data.requested_objects.list
      + $rel[0].data.requested_objects.list
    ) as $scope

  | (reduce $all[] as $o ({}; .[$o.name] = $o)) as $byName

  | ([$scope[].id | tostring] | unique) as $scopeIds

  | [
      $scope[]
      | . as $parent
      | tokens($parent.ipAddresses)[]
      | select(startswith("grp"))
      | . as $name
      | $byName[$name] as $child
      | select($child != null)
      | select(($scopeIds | index($child.id | tostring)) == null)
      | $child
    ] | unique_by(.id) as $missing

  | [
      $missing[]
      | . as $parent
      | tokens($parent.ipAddresses)[]
      | select(startswith("grp"))
      | . as $name
      | $byName[$name] as $child
      | {
          parentId: $parent.id,
          parentName: $parent.name,
          childId: ($child.id // null),
          childName: $name,
          childEonid: ($child.eonid // null),
          existsInRaw: ($child != null),
          alreadyInScope: (
            if $child == null then false
            else ($scopeIds | index($child.id | tostring)) != null
            end
          )
        }
    ] as $next

  | {
      currentMissingGroups: ($missing | length),

      nextLevelReferences: ($next | length),

      nextLevelUniqueGroups: (
        [$next[].childId | select(. != null) | tostring]
        | unique | length
      ),

      unresolvedGroupReferences: (
        [$next[] | select(.existsInRaw | not)]
        | length
      ),

      newDependencies: [
        $next[]
        | select(
            .existsInRaw
            and (.alreadyInScope | not)
          )
      ],

      examples: $next[:10]
    }
'
