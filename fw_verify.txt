jq -n \
  --slurpfile r filter/rules.json \
  --slurpfile f filter/flows.json '
  ([$f[0].data.requested_objects.list[].flowid | tostring]
   | unique) as $ids

  | [
      $r[0].data.requested_rules.list[]
      | select((.flowdbIds // "") | contains(","))
      | {
          ruleId: .id,
          flowIds: (
            .flowdbIds
            | split(",")
            | map(gsub("^\\s+|\\s+$"; ""))
          )
        }
      | . + {
          flowCount: (.flowIds | length),
          missingFlowIds: [
            .flowIds[]
            | select(. as $id | $ids | index($id) == null)
          ]
        }
    ]
'

jq -n \
  --slurpfile f filter/flows.json '
  def norm:
    if . == null then ""
    else tostring
    end;

  (
    [$f[0].data.requested_objects.list[].flowid | norm]
    | unique
  ) as $ids

  | [
      $f[0].data.requested_objects.list[]
      | (.oldFlowid | norm) as $prev
      | select(
          $prev != ""
          and $prev != "0"
          and ($ids | index($prev)) == null
        )
      | {
          flowid,
          oldFlowid,
          sourceEonid,
          destinationEonid
        }
    ]
'

jq --arg id "这里替换成oldFlowid" '
  .data.requested_objects.list[]
  | select((.flowid | tostring) == $id)
  | {
      flowid,
      oldFlowid,
      sourceEonid,
      destinationEonid,
      sourceSecurityZone,
      destinationSecurityZone
    }
' flows.json
