jq -n \
  --slurpfile r filter/rules.json \
  --slurpfile o objects.json '
  def names:
    (. // "")
    | split(",")
    | map(gsub("^\\s+|\\s+$"; ""))
    | map(select(length > 0))[];

  ([
    $r[0].data.requested_rules.list[]
    | (.sourceObjectDetails, .destinationObjectDetails)
    | names
  ] | unique) as $wanted

  | [
      $o[0].data.requested_objects.list[]
      | select(.name as $n | $wanted | index($n) != null)
      | select(.type == "DECO" or .status != "Active")
      | {
          id,
          name,
          eonid,
          status,
          type,
          isDeleted
        }
    ]
'
