jq -n \
  --slurpfile raw objects.json \
  --slurpfile p filter/objects.json \
  --slurpfile rel filter/related_objects.json '
  def tokens($v):
    (($v // "") | tostring
    | split(",")
    | map(gsub("^\\s+|\\s+$"; ""))
    | map(select(length > 0)));

  ($raw[0].data.requested_objects.list) as $all

  | (
      $p[0].data.requested_objects.list
      + $rel[0].data.requested_objects.list
    ) as $scope

  | (reduce $all[] as $o ({}; .[$o.name] = $o)) as $byName

  | ([$scope[].id | tostring] | unique) as $scopeIds

  | [
      $scope[] as $parent
      | tokens($parent.ipAddresses)[]
      | select(startswith("grp"))
      | . as $name
      | $byName[$name] as $child
      | select($child != null)
      | {
          parentId: $parent.id,
          childId: $child.id
        }
    ] as $edges1

  | ([$edges1[].childId | tostring] | unique) as $childIds

  | [
      $all[]
      | select(
          (.id | tostring) as $id
          | $childIds | index($id) != null
        )
    ] as $level1

  | [
      $level1[] as $parent
      | tokens($parent.ipAddresses)[]
      | select(startswith("grp"))
      | . as $name
      | $byName[$name] as $child
      | {
          parentId: $parent.id,
          parentName: $parent.name,
          childName: $name,
          childId: ($child.id // null),
          existsInRaw: ($child != null),
          childInScope: (
            if $child == null then false
            else ($scopeIds | index($child.id | tostring)) != null
            end
          )
        }
    ] as $edges2

  | {
      firstLevelChildGroups: ($level1 | length),
      secondLevelReferenceInstances: ($edges2 | length),
      secondLevelUniqueChildren: (
        [$edges2[].childId | select(. != null) | tostring]
        | unique | length
      ),
      unresolvedReferences: (
        [$edges2[] | select(.existsInRaw | not)]
        | length
      ),
      additionalDependencies: (
        [$edges2[]
         | select(.existsInRaw and (.childInScope | not))
         | .childId | tostring]
        | unique | length
      ),
      examples: $edges2[:10]
    }
'
