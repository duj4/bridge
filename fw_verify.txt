jq -n \
  --slurpfile raw raw/objects.json \
  --slurpfile p filter/objects.json \
  --slurpfile rel filter/related_objects.json \
  --slurpfile nest filter/nested_objects.json \
  --slurpfile e graph/edge.json '
  def tokens:
    (. // "" | tostring
    | split(",")[]
    | gsub("^\\s+|\\s+$"; "")
    | select(length > 0));

  (
    $p[0].data.requested_objects.list
    + $rel[0].data.requested_objects.list
    + $nest[0].data.requested_objects.list
  ) as $scope

  | (
    reduce $raw[0].data.requested_objects.list[] as $o
      ({}; .[$o.name] = $o.id)
  ) as $byName

  | [
      $scope[] as $parent
      | ($parent.ipAddresses | tokens) as $token
      | $byName[$token] as $child
      | select($child != null)
      | {
          source: "group:\($parent.id)",
          target: "group:\($child)",
          type: "GROUP_CONTAINS_GROUP"
        }
    ] | sort as $expected

  | [
      $e[0].edges[]
      | select(.type == "GROUP_CONTAINS_GROUP")
      | {source, target, type}
    ] | sort as $actual

  | {
      expectedNestedEdges: ($expected | length),
      actualNestedEdges: ($actual | length),
      exactMatch: ($expected == $actual)
    }
'

jq -n \
  --slurpfile f filter/flows.json \
  --slurpfile e graph/edge.json '
  ($f[0].data.requested_objects.list) as $flows

  | (
      reduce $flows[] as $x
        ({}; .[($x.flowid | tostring)] = true)
    ) as $idx

  | [
      $flows[] as $x
      | ($x.oldFlowid // "" | tostring) as $old
      | select(
          $old != ""
          and $old != "0"
          and ($idx[$old] // false)
        )
      | {
          source: "flow:\($x.flowid)",
          target: "flow:\($old)",
          type: "FLOW_PREVIOUS_VERSION"
        }
    ] | sort as $expected

  | [
      $e[0].edges[]
      | select(.type == "FLOW_PREVIOUS_VERSION")
      | {source, target, type}
    ] | sort as $actual

  | {
      expectedEvolutionEdges: ($expected | length),
      actualEvolutionEdges: ($actual | length),
      exactMatch: ($expected == $actual)
    }
'
