cat > flb-heap-diagnostic.sh <<'SCRIPT'
#!/usr/bin/env bash
# Fluent Bit 5.1.5 Heap Profiling Diagnostics
# Heap files remain on OpenShift workers.
# Only diagnostic text reports are saved on the jump host.

set -Eeuo pipefail
umask 077

# ============================================================
# Configuration
# ============================================================

NS="${NS:-}"
LOG_SINCE="${LOG_SINCE:-12h}"

PROFILE_DIR="${PROFILE_DIR:-/var/log/fluent-bit-heap-profiles}"
OUT_DIR="${OUT_DIR:-$HOME/flb-heap-reports}"

# 0 = read-only diagnostics
# 1 = additionally run jeprof when sufficient memory is available
ANALYZE="${ANALYZE:-0}"

# Minimum free memory required before running jeprof inside a live pod
MIN_HEADROOM_MIB="${MIN_HEADROOM_MIB:-192}"

command -v kubectl >/dev/null || {
    echo "ERROR: kubectl not found" >&2
    exit 1
}

[[ "$ANALYZE" =~ ^[01]$ ]] || {
    echo "ERROR: ANALYZE must be 0 or 1" >&2
    exit 1
}

[[ "$MIN_HEADROOM_MIB" =~ ^[0-9]+$ ]] || {
    echo "ERROR: MIN_HEADROOM_MIB must be numeric" >&2
    exit 1
}

K=(kubectl)
[[ -z "$NS" ]] || K+=(-n "$NS")

# ============================================================
# Discover Fluent Bit Pods
# ============================================================

if (( $# > 0 )); then
    PODS=("$@")
else
    mapfile -t PODS < <(
        "${K[@]}" get pods \
            -o jsonpath='{range .items[*]}{.metadata.name}{"\t"}{.status.phase}{"\n"}{end}' |
        awk '$1 ~ /^fluent-bit-[a-z0-9]+$/ && $2 == "Running" {print $1}'
    )
fi

(( ${#PODS[@]} > 0 )) || {
    echo "ERROR: No running Fluent Bit Pods found" >&2
    exit 1
}

# ============================================================
# Report Initialization
# ============================================================

STAMP="$(date -u '+%Y%m%dT%H%M%SZ')"

mkdir -p "$OUT_DIR"

REPORT="$OUT_DIR/flb-heap-$STAMP.txt"

{
    echo "Report UTC: $(date -u '+%Y-%m-%dT%H:%M:%SZ')"

    echo "kubectl context: $("${K[@]}" config current-context 2>/dev/null || echo unknown)"

    echo "Namespace: ${NS:-<current>}"
    echo "Log lookback: $LOG_SINCE"
    echo "Profile directory: $PROFILE_DIR"
    echo "jeprof analysis: $ANALYZE"
    echo "Minimum headroom: ${MIN_HEADROOM_MIB} MiB"
    echo "Pods (${#PODS[@]}): ${PODS[*]}"

    # ========================================================
    # Process each Pod
    # ========================================================

    for pod in "${PODS[@]}"; do

        echo
        echo "==================== $pod ===================="

        # ----------------------------------------------------
        # Container Identity
        # ----------------------------------------------------

        echo "=== POD / CONTAINERS ==="

        if ! "${K[@]}" get pod "$pod" \
            -o jsonpath='{.spec.nodeName}{"\n"}{range .status.containerStatuses[*]}{.name}{" restarts="}{.restartCount}{" startedAt="}{.state.running.startedAt}{" lastReason="}{.lastState.terminated.reason}{" lastExit="}{.lastState.terminated.exitCode}{" image="}{.image}{"\n"}{end}'; then

            echo "ERROR: Cannot read Pod information"
            continue
        fi

        echo

        STARTED="$(
            "${K[@]}" get pod "$pod" \
                -o jsonpath='{range .status.containerStatuses[?(@.name=="fluent-bit")]}{.state.running.startedAt}{end}' \
                2>/dev/null || true
        )"

        if [[ -n "$STARTED" ]]; then
            START_EPOCH="$(
                date -u -d "$STARTED" +%s 2>/dev/null || echo 0
            )"
        else
            START_EPOCH=0
            echo "WARN: Cannot identify Fluent Bit startup time"
        fi

        # ----------------------------------------------------
        # jemalloc Statistics
        # ----------------------------------------------------

        echo "=== JEMALLOC STATS ==="

        STATS="$(
            "${K[@]}" logs "$pod" \
                -c fluent-bit \
                --since="$LOG_SINCE" \
                --timestamps 2>/dev/null |
                grep -F 'Allocated:' |
                tail -3 || true
        )"

        if [[ -n "$STATS" ]]; then
            printf '%s\n' "$STATS"
        else
            echo "No Allocated lines found."
            echo "This is expected if stats_interval is not configured."
        fi

        # ----------------------------------------------------
        # Profiling Errors
        # ----------------------------------------------------

        echo "=== PROFILING / MALLOC ERROR LOGS ==="

        ERRORS="$(
            "${K[@]}" logs "$pod" \
                -c fluent-bit \
                --since="$LOG_SINCE" \
                --timestamps 2>/dev/null |
                grep -Ei 'jemalloc.*(invalid|malformed|failed)|failed to (open|write).*heap|segfault|signal 11|signal 139|out of memory' |
                tail -12 || true
        )"

        if [[ -n "$ERRORS" ]]; then
            printf '%s\n' "$ERRORS"
        else
            echo "No matching errors found."
        fi

        # ----------------------------------------------------
        # Live Process / CGROUP / Heap Profiles
        # ----------------------------------------------------

        echo "=== LIVE PROCESS / CGROUP / HEAP ==="

        if ! "${K[@]}" exec -i "$pod" -c fluent-bit -- \
            sh -s -- \
            "$pod" \
            "$PROFILE_DIR" \
            "$START_EPOCH" \
            "$ANALYZE" \
            "$MIN_HEADROOM_MIB" <<'IN_POD'

POD="$1"
PROF_DIR="$2"
START="$3"
ANALYZE="$4"
MIN_HEADROOM="$5"

printf 'Sample UTC: '
date -u '+%Y-%m-%dT%H:%M:%SZ'

# ============================================================
# PID 1 jemalloc Configuration
# ============================================================

echo "--- PID 1 MALLOC_CONF ---"

MALLOC_CONF_TEXT="$(
    tr '\000' '\n' < /proc/1/environ |
        grep '^MALLOC_CONF=' |
        head -1
)"

if [ -n "$MALLOC_CONF_TEXT" ]; then
    printf '%s\n' "$MALLOC_CONF_TEXT"
else
    echo "WARN: MALLOC_CONF not found"
fi

case "$MALLOC_CONF_TEXT" in
    *'prof:true'*'prof_active:true'*)
        echo "Profiling appears configured as active."
        ;;
    *)
        echo "WARN: Active profiling configuration not confirmed."
        ;;
esac

case "$MALLOC_CONF_TEXT" in
    *'stats_interval:'*)
        echo "Periodic jemalloc Statistics enabled."
        ;;
    *)
        echo "Periodic jemalloc Statistics not configured."
        ;;
esac

# ============================================================
# Process Memory
# ============================================================

echo "--- PROCESS STATUS ---"

grep -E '^(VmRSS|RssAnon|Threads):' \
    /proc/1/status || true

echo "--- PROCESS SMAPS ROLLUP ---"

grep -E '^(Rss|Pss|Private_Dirty|Anonymous|AnonHugePages|LazyFree):' \
    /proc/1/smaps_rollup || true

# ============================================================
# File Descriptors / Storage
# ============================================================

echo "--- FD / STORAGE ---"

printf 'FD count: '
ls -1 /proc/1/fd 2>/dev/null | wc -l

printf 'Fluent Bit storage: '

du -sh /var/log/fluent-bit-storage 2>/dev/null ||
    echo "N/A"

# ============================================================
# Fluent Bit Metrics Health
# ============================================================

echo "--- METRICS HTTP SERVER ---"

if command -v curl >/dev/null 2>&1; then

    if curl -fsS --max-time 5 \
        http://127.0.0.1:2020/api/v2/metrics/prometheus \
        >/dev/null 2>&1; then

        echo "Metrics endpoint: HTTP OK"

    else
        echo "WARN: Metrics endpoint unavailable"
    fi

else
    echo "curl unavailable; metrics check skipped"
fi

# ============================================================
# Container cgroup
# ============================================================

CG_REL="$(sed -n 's/^0:://p' /proc/1/cgroup)"
CG="/sys/fs/cgroup${CG_REL}"

echo "--- CONTAINER CGROUP ---"
echo "Cgroup path: $CG"

HEADROOM_BYTES=0

if [ -r "$CG/memory.current" ] &&
   [ -r "$CG/memory.max" ]; then

    CUR="$(cat "$CG/memory.current")"
    MAX="$(cat "$CG/memory.max")"

    if [ "$MAX" != max ]; then

        HEADROOM_BYTES=$((MAX - CUR))

        awk -v c="$CUR" -v m="$MAX" '
        BEGIN {
            printf "Current: %.1f MiB\n", c/1048576
            printf "Limit: %.1f MiB\n", m/1048576
            printf "Headroom: %.1f MiB\n", (m-c)/1048576
        }'

    else
        echo "Current: $CUR bytes"
        echo "Limit: max"
    fi

    if [ -r "$CG/memory.peak" ]; then
        printf 'Peak (bytes): '
        cat "$CG/memory.peak"
    fi

    echo "--- MEMORY.STAT ---"
    echo "(Bytes; some fields overlap)"

    grep -E '^(anon|anon_thp|file|kernel|slab|slab_reclaimable|slab_unreclaimable|inactive_file|active_file) ' \
        "$CG/memory.stat" || true

    echo "--- MEMORY EVENTS ---"

    cat "$CG/memory.events" 2>/dev/null || true

else
    echo "WARN: Container cgroup not resolved"
fi

# ============================================================
# Heap Profile Directory
# ============================================================

echo "--- HEAP PROFILE DIRECTORY ---"

if [ ! -d "$PROF_DIR" ]; then
    echo "ERROR: Profile directory does not exist"
    echo "$PROF_DIR"
    exit 0
fi

ls -ld "$PROF_DIR" 2>/dev/null || true

echo "--- PROFILE DISK USAGE ---"

du -sh "$PROF_DIR" 2>/dev/null || true

df -h "$PROF_DIR" 2>/dev/null | tail -2 || true

# ============================================================
# Discover profiles for CURRENT container lifecycle
# ============================================================

PROFILES="$(
    find "$PROF_DIR" \
        -maxdepth 1 \
        -type f \
        -name "${POD}.*.i*.heap" \
        -printf '%T@ %p\n' 2>/dev/null |
    awk -v since="$START" '
        $1 + 0 >= since + 0 {
            print $2
        }
    ' |
    sort -V
)"

COUNT="$(
    printf '%s\n' "$PROFILES" |
        awk 'NF {n++} END {print n+0}'
)"

echo "--- HEAP PROFILE SUMMARY ---"
echo "Current-process interval heap files: $COUNT"

if [ "$COUNT" -eq 0 ]; then

    echo "No interval Heap Dump found for current process."

    echo "With lg_prof_interval:40, a dump is triggered"
    echo "after approximately 1 TiB cumulative allocations."
    echo "This is not a time-based interval."

    exit 0
fi

FIRST="$(
    printf '%s\n' "$PROFILES" | head -1
)"

LAST="$(
    printf '%s\n' "$PROFILES" | tail -1
)"

echo "First Profile: $FIRST"
echo "Last Profile:  $LAST"

echo "--- LATEST 5 HEAP FILES ---"

printf '%s\n' "$PROFILES" |
    tail -5 |
    while IFS= read -r p; do
        [ -f "$p" ] &&
            stat -c '%y  %s bytes  %n' "$p" || true
    done

# ============================================================
# Optional jeprof Analysis
# ============================================================

if [ "$ANALYZE" != 1 ]; then
    echo "jeprof analysis disabled (ANALYZE=0)."
    exit 0
fi

# jeprof uses additional memory in the SAME container cgroup.
# Do not run if there is insufficient headroom.

MIN_BYTES=$((MIN_HEADROOM * 1048576))

if [ "$HEADROOM_BYTES" -lt "$MIN_BYTES" ]; then

    echo "SKIP jeprof: insufficient memory headroom."
    echo "Required: ${MIN_HEADROOM} MiB"

    exit 0
fi

if ! command -v jeprof >/dev/null 2>&1; then

    echo "SKIP jeprof: binary not installed"
    exit 0
fi

FILTER='^(prof_backtrace_impl|je_prof_backtrace|je_prof_tctx_create|je_malloc_default|prof_alloc_prep)$'

BIN=/fluent-bit/bin/fluent-bit

# ============================================================
# Latest Heap
# ============================================================

echo "--- JEPROF LATEST IN-USE ALLOCATIONS ---"

timeout 90 jeprof \
    --text \
    --show_bytes \
    --exclude="$FILTER" \
    "$BIN" \
    "$LAST" 2>&1 | head -35 || true

# ============================================================
# Heap Difference
# ============================================================

if [ "$COUNT" -ge 2 ] &&
   [ "$FIRST" != "$LAST" ]; then

    echo "--- JEPROF DIFF: LAST - FIRST ---"

    timeout 90 jeprof \
        --text \
        --show_bytes \
        --exclude="$FILTER" \
        --base="$FIRST" \
        "$BIN" \
        "$LAST" 2>&1 | head -40 || true

else
    echo "Only one valid Heap Profile; Diff unavailable."
fi

IN_POD
        then
            echo "WARN: Diagnostic exec failed for $pod"
            echo "Remaining Pods will still be checked."
        fi

    done

} 2>&1 | tee "$REPORT"

echo
echo "Saved report: $REPORT"

SCRIPT
