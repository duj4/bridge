cat > flb-jeprof-isolated.sh <<'SCRIPT'
#!/usr/bin/env bash
# Fluent Bit jeprof analyzer:
# No hostPath, no node SSH, no DaemonSet changes.

set -Eeuo pipefail
umask 077

NS="${NS:-}"
OUT_DIR="${OUT_DIR:-$HOME/flb-heap-reports}"
PROFILE_DIR="${PROFILE_DIR:-/var/log/fluent-bit-heap-profiles}"
MAX_PROFILES="${MAX_PROFILES:-48}"
KEEP_POD="${KEEP_POD:-0}"
WAIT_SECONDS="${WAIT_SECONDS:-300}"

K=(kubectl)
[[ -z "$NS" ]] || K+=(-n "$NS")

fail() {
  echo "ERROR: $*" >&2
  exit 1
}

for cmd in kubectl awk sort date tar; do
  command -v "$cmd" >/dev/null 2>&1 ||
    fail "Missing command: $cmd"
done

[[ "$MAX_PROFILES" =~ ^[0-9]+$ ]] &&
  ((MAX_PROFILES >= 3 && MAX_PROFILES <= 500)) ||
  fail "MAX_PROFILES must be 3..500"

[[ "$WAIT_SECONDS" =~ ^[0-9]+$ ]] &&
  ((WAIT_SECONDS >= 30)) ||
  fail "WAIT_SECONDS must be >=30"

[[ "$KEEP_POD" == 0 || "$KEEP_POD" == 1 ]] ||
  fail "KEEP_POD must be 0 or 1"

mkdir -p "$OUT_DIR"

# ============================================================
# 1. Discover Running Fluent Bit Pods
# ============================================================

if (($# == 0)) || [[ "${1:-}" == all ]]; then

  mapfile -t TARGETS < <(
    "${K[@]}" get pods \
      -o jsonpath='{range .items[*]}{.metadata.name}{"\t"}{.status.phase}{"\n"}{end}' |
      awk '$1 ~ /^fluent-bit-[a-z0-9]+$/ && $2=="Running" {print $1}' |
      sort
  )

else
  TARGETS=("$@")
fi

((${#TARGETS[@]} > 0)) ||
  fail "No Running Fluent Bit Pods discovered"

# ============================================================
# 2. Analyze One Fluent Bit Pod
# ============================================================

run_one() (
  set -Eeuo pipefail

  local target="$1"

  [[ "$target" =~ ^fluent-bit-[a-z0-9]+$ ]] ||
    fail "Invalid Pod name: $target"

  local stamp report manifest analyzer='' created=0

  stamp="$(date -u +%Y%m%dT%H%M%SZ)"

  report="$OUT_DIR/flb-jeprof-${target}-${stamp}-$$.txt"
  manifest="$OUT_DIR/flb-jeprof-${target}-${stamp}-$$.yaml"

  exec > >(tee -a "$report") 2>&1

  # ----------------------------------------------------------
  # Automatic Cleanup / Failure Diagnostics
  # ----------------------------------------------------------

  cleanup() {
    local rc=$?
    trap - EXIT

    if ((created)); then

      if ((rc == 0)) && [[ "$KEEP_POD" != 1 ]]; then

        "${K[@]}" delete pod "$analyzer" \
          --wait=false \
          --ignore-not-found=true || true

      else

        if ((rc != 0)); then
          echo "=== ANALYZER FAILURE DIAGNOSTICS ==="

          "${K[@]}" get pod "$analyzer" -o wide || true

          "${K[@]}" logs "$analyzer" \
            -c analyzer --tail=50 2>&1 || true

          "${K[@]}" describe pod "$analyzer" 2>&1 |
            sed -n '/Events:/,$p' || true
        fi

        echo "Analyzer retained: $analyzer"
        echo "Cleanup when finished: kubectl ${NS:+-n $NS }delete pod $analyzer"
      fi
    fi

    echo "Report: $report"
    exit "$rc"
  }

  trap cleanup EXIT

  # ==========================================================
  # 3. Read Target Container Identity
  # ==========================================================

  local node sa image image_id digest_image
  local started start_epoch cid pod_uid

  node="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.spec.nodeName}')"

  sa="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.spec.serviceAccountName}')"

  image="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.spec.containers[?(@.name=="fluent-bit")].image}')"

  image_id="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.status.containerStatuses[?(@.name=="fluent-bit")].imageID}')"

  started="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.status.containerStatuses[?(@.name=="fluent-bit")].state.running.startedAt}')"

  cid="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.status.containerStatuses[?(@.name=="fluent-bit")].containerID}')"

  pod_uid="$("${K[@]}" get pod "$target" \
    -o jsonpath='{.metadata.uid}')"

  [[ -n "$node" && -n "$image" &&
     -n "$started" && -n "$cid" &&
     -n "$pod_uid" ]] ||
    fail "Fluent Bit is not Running or metadata unavailable"

  [[ "$image" == *heap-prof-libunwind* ]] ||
    fail "Unexpected image; a matching jeprof-enabled image is required: $image"

  sa="${sa:-default}"
  start_epoch="$(date -u -d "$started" +%s)"

  # Prefer the exact image digest for symbol matching.

  digest_image="${image_id#docker-pullable://}"
  digest_image="${digest_image#docker://}"

  if [[ "$digest_image" =~ ^[^[:space:]]+@sha256:[[:xdigit:]]{64}$ ]]; then
    image="$digest_image"
  fi

  analyzer="flb-jp-${target#fluent-bit-}-$(date -u +%m%d%H%M%S)-${BASHPID}"

  echo "=== SOURCE IDENTITY ==="
  echo "Pod: $target   Node: $node   UID: $pod_uid"
  echo "Started: $started   Container ID: $cid"
  echo "Analyzer image: $image"
  echo "Profile directory: $PROFILE_DIR"

  # ==========================================================
  # 4. Discover Heap Files from Current Container Lifecycle
  # ==========================================================

  local raw file n i idx first middle last count total

  raw="$(
    "${K[@]}" exec "$target" -c fluent-bit -- \
      find "$PROFILE_DIR" \
        -maxdepth 1 \
        -type f \
        -name "${target}.*.i*.heap" \
        -mmin +1 \
        -size +0c \
        -printf '%T@ %f\n'
  )"

  mapfile -t CANDIDATES < <(
    printf '%s\n' "$raw" |
      awk -v t="$start_epoch" '$1+0 >= t+0 {print $2}' |
      sort -V
  )

  total="${#CANDIDATES[@]}"

  ((total >= 2)) ||
    fail "Need at least two stable .i.heap profiles from current process (found $total)"

  for file in "${CANDIDATES[@]}"; do

    [[ "$file" =~ ^${target}\.[0-9]+\.[0-9]+\.i[0-9]+\.heap$ ]] ||
      fail "Unexpected file name: $file"

  done

  # ==========================================================
  # 5. Select Profiles
  #
  # If there are too many, sample evenly over the entire
  # process lifetime. Always include first and last.
  # ==========================================================

  SELECTED=()

  if ((total <= MAX_PROFILES)); then

    SELECTED=("${CANDIDATES[@]}")

  else

    for ((i=0; i<MAX_PROFILES; i++)); do

      idx=$((i*(total-1)/(MAX_PROFILES-1)))
      SELECTED+=("${CANDIDATES[idx]}")

    done
  fi

  count="${#SELECTED[@]}"

  first="${SELECTED[0]}"
  middle="${SELECTED[count/2]}"
  last="${SELECTED[count-1]}"

  echo "Found $total complete dumps; analyzing $count evenly spaced dumps."
  echo "First=$first"
  echo "Middle=$middle"
  echo "Last=$last"

  # ==========================================================
  # 6. Create Isolated Analysis Pod
  #
  # NO HostPath.
  # NO Privileged.
  # Independent 1 GiB memory limit.
  # ==========================================================

  cat > "$manifest" <<YAML
apiVersion: v1
kind: Pod
metadata:
  name: "$analyzer"
  labels:
    app: flb-jeprof-analysis
spec:
  serviceAccountName: "$sa"
  automountServiceAccountToken: false
  restartPolicy: Never

  containers:
    - name: analyzer
      image: "$image"
      imagePullPolicy: IfNotPresent

      command: ["/bin/sh", "-ec"]
      args: ["mkdir -p /work/profiles /work/tmp && echo ANALYZER_READY && sleep 7200"]

      securityContext:
        runAsNonRoot: true
        readOnlyRootFilesystem: true
        allowPrivilegeEscalation: false
        seccompProfile:
          type: RuntimeDefault
        capabilities:
          drop: ["ALL"]

      resources:
        requests:
          cpu: 100m
          memory: 256Mi
        limits:
          cpu: 500m
          memory: 1Gi

      volumeMounts:
        - name: work
          mountPath: /work

  volumes:
    - name: work
      emptyDir:
        sizeLimit: 512Mi
YAML

  echo "=== CREATE RESTRICTED ANALYZER (NO hostPath) ==="

  "${K[@]}" create -f "$manifest"
  created=1

  "${K[@]}" wait \
    --for=condition=Ready \
    "pod/$analyzer" \
    --timeout="${WAIT_SECONDS}s"

  "${K[@]}" exec "$analyzer" -c analyzer -- \
    sh -ec \
      'test -w /work && command -v jeprof && command -v tar && test -r /fluent-bit/bin/fluent-bit'

  # ==========================================================
  # 7. Transfer Heap Files Between Pods
  #
  # Original .heap data stays inside company infrastructure.
  # No files are saved on jump-host disk.
  # ==========================================================

  echo "=== STREAM $count HEAP FILES ==="

  "${K[@]}" exec "$target" -c fluent-bit -- \
    tar -C "$PROFILE_DIR" -cf - -- "${SELECTED[@]}" |
    "${K[@]}" exec -i "$analyzer" -c analyzer -- \
      tar -C /work/profiles -xf -

  # Check that the source container did not restart.

  [[ "$("${K[@]}" get pod "$target" \
    -o jsonpath='{.status.containerStatuses[?(@.name=="fluent-bit")].containerID}')" == "$cid" ]] ||
    fail "Source container restarted during transfer"

  # ==========================================================
  # 8. Execute Heap Analysis
  # ==========================================================

  echo "=== ANALYZE ==="

  "${K[@]}" exec -i "$analyzer" -c analyzer -- \
    env \
      TARGET_POD="$target" \
      FIRST="$first" \
      MIDDLE="$middle" \
      LAST="$last" \
      HOME=/work \
      TMPDIR=/work/tmp \
      JEPROF_TMPDIR=/work/tmp \
      sh -s <<'IN_ANALYZER'

set -eu

BIN=/fluent-bit/bin/fluent-bit

FILTER='^(prof_backtrace_impl|je_prof_backtrace|je_prof_tctx_create|je_malloc_default|prof_alloc_prep)$'

cd /work/profiles

for f in "$FIRST" "$MIDDLE" "$LAST"; do
  test -s "$f" || {
    echo "ERROR: missing profile $f"
    exit 21
  }
done

# ============================================================
# 9. Heap Time Series
# ============================================================

find . -maxdepth 1 -type f \
  -name "${TARGET_POD}.*.i*.heap" \
  -printf '%f\n' |
  sort -V > /work/selected-files.txt

printf '\n=== HEAP TIME SERIES (sampled estimated in-use MiB) ===\n'

printf 'FILE\tUTC\tTOTAL\tLOKI_PATH\tTAIL\tMETRICS\tCHUNK\tOTHER\n'

while IFS= read -r file; do

  [ -n "$file" ] || continue

  utc="$(
    date -u \
      -d "@$(stat -c %Y "$file")" \
      '+%Y-%m-%dT%H:%M:%SZ'
  )"

  if ! timeout 150 jeprof \
    --collapsed \
    --show_bytes \
    --inuse_space \
    --exclude="$FILTER" \
    "$BIN" "$file" \
    > /work/current.collapsed \
    2> /work/jeprof.err; then

    echo "ERROR: jeprof failed for $file"
    cat /work/jeprof.err
    exit 22
  fi

  awk -v f="$file" -v utc="$utc" '
    $NF ~ /^[0-9]+$/ && NF>=2 {

      v=$NF+0
      all+=v

      if (
        index($0,"cb_loki_flush") ||
        index($0,"send_loki_payload") ||
        index($0,"loki_compose_payload")
      ) {
        loki+=v
      }

      else if (
        index($0,"flb_tail_") ||
        index($0,"in_tail_") ||
        index($0,"tail_fs_event")
      ) {
        tail+=v
      }

      else if (
        index($0,"cb_metrics_prometheus") ||
        index($0,"unpack_metric_array_entry") ||
        index($0,"cmt_") ||
        index($0,"flb_hs_push")
      ) {
        metrics+=v
      }

      else if (
        index($0,"cio_memfs_open") ||
        index($0,"flb_input_chunk_create")
      ) {
        chunk+=v
      }

      else {
        other+=v
      }
    }

    END {
      unit=1048576

      printf "%s\t%s\t%.2f\t%.2f\t%.2f\t%.2f\t%.2f\t%.2f\n",
        f,utc,all/unit,loki/unit,tail/unit,
        metrics/unit,chunk/unit,other/unit
    }
  ' /work/current.collapsed

done < /work/selected-files.txt

echo '=== END HEAP TIME SERIES ==='

# ============================================================
# 10. Growth Analysis
#
# First -> Middle
# Middle -> Last
# First -> Last
# ============================================================

show_delta() {

  label="$1"
  base="$2"
  current="$3"

  echo
  echo "=== HEAP GROWTH: $label ==="

  if ! timeout 180 jeprof \
    --collapsed \
    --show_bytes \
    --inuse_space \
    --exclude="$FILTER" \
    --base="$base" \
    "$BIN" "$current" \
    > /work/delta.collapsed \
    2> /work/jeprof.err; then

    echo "ERROR: growth analysis failed"
    cat /work/jeprof.err
    exit 23
  fi

  awk '
    $NF ~ /^-?[0-9]+$/ {
      s+=$NF
    }

    END {
      printf "Sampled in-use net difference: %+.2f MiB\n",
        s/1048576
    }
  ' /work/delta.collapsed

  # ----------------------------------------------------------
  # Compact Full Stack Paths
  # ----------------------------------------------------------

  for mode in POSITIVE NEGATIVE; do

    echo "--- TOP $mode FULL PATHS (estimated bytes) ---"

    awk -v mode="$mode" '
      NF>=2 && $NF ~ /^-?[0-9]+$/ {

        value=$NF+0

        if (
          (mode=="POSITIVE" && value<=0) ||
          (mode=="NEGATIVE" && value>=0)
        ) next

        line=$0

        sub(/[[:space:]]+-?[0-9]+$/, "", line)

        n=split(line,frames,";")
        out=""

        for (i=1;i<=n;i++) {

          frame=frames[i]

          sub(/<[^>]*>$/, "", frame)

          if (frame ~ /flb_|in_tail|tail_fs_event|cb_|send_loki|cio_|msgpack|CRYPTO_|SSL_|cmt_|unpack_|encoder_|rallocx|sds_|_create/) {

            out=out (out=="" ? "" : " > ") frame

          }
        }

        if (out=="") {
          out=substr(line,1,240)
        }

        printf "%.0f\t%s\n",
          value,substr(out,1,650)
      }
    ' /work/delta.collapsed |

      if [ "$mode" = POSITIVE ]; then
        sort -k1,1nr | sed -n '1,12p'
      else
        sort -k1,1n | sed -n '1,8p'
      fi

  done
}

show_delta \
  'FIRST -> MIDDLE' \
  "$FIRST" "$MIDDLE"

show_delta \
  'MIDDLE -> LAST' \
  "$MIDDLE" "$LAST"

show_delta \
  'FIRST -> LAST' \
  "$FIRST" "$LAST"

echo '=== ANALYSIS COMPLETE ==='

IN_ANALYZER

  # ==========================================================
  # 11. Final Container Identity Check
  # ==========================================================

  local latest_cid

  latest_cid="$(
    "${K[@]}" get pod "$target" \
      -o jsonpath='{.status.containerStatuses[?(@.name=="fluent-bit")].containerID}'
  )"

  [[ "$latest_cid" == "$cid" ]] ||
    fail "Source container restarted during analysis; results invalid"

  echo "ANALYSIS SUCCEEDED: $target"
)

# ============================================================
# 12. Process Pods Sequentially
# ============================================================

for pod in "${TARGETS[@]}"; do
  run_one "$pod"
done

echo "All requested Pod analyses completed. Reports: $OUT_DIR"

SCRIPT

chmod +x flb-jeprof-isolated.sh
bash -n flb-jeprof-isolated.sh && echo "BASH SYNTAX PASSED"
