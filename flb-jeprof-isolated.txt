cp -p flb-jeprof-isolated.sh \
  "flb-jeprof-isolated.sh.bak.$(date +%Y%m%d%H%M%S)"

cat > /tmp/flb-jeprof-v7-patch.py <<'PY'
from pathlib import Path
import os
import shutil
import subprocess
import sys

p = Path("flb-jeprof-isolated.sh")
s = p.read_text()

old = 'echo "FLB_JEPROF_AWK_V6_20261102"'
new = 'echo "FLB_JEPROF_V7_PERL_FULLSTACK_20261102"'
marker = "echo 'AWK CLASSIFIER SELFTEST PASSED'\n"
begin = "show_delta() {"
end = "show_delta 'FIRST -> MIDDLE'"

if new in s:
    sys.exit("Already V7")

if any(s.count(x) != 1 for x in (old, marker, begin, end)):
    sys.exit("ERROR: Unexpected V6 script structure")

setup = r'''
cat > /work/stack-top.pl <<'PERL_V7'
use strict;
use warnings;

my ($file, $mode, $limit) = @ARGV;

die "Invalid mode\n"
    unless $mode eq "POSITIVE" || $mode eq "NEGATIVE";

open(my $fh, "<", $file)
    or die "Cannot open $file: $!\n";

my @rows;
my $nonempty = 0;
my $parsed = 0;

while (my $line = <$fh>) {
    next if $line =~ /^\s*$/;
    $nonempty++;

    next unless $line =~ /^\s*(.*?)\s+([+-]?\d+)\s*$/;

    my ($stack, $bytes) = ($1, 0 + $2);
    $parsed++;

    next if $bytes == 0;
    next if $mode eq "POSITIVE" && $bytes < 0;
    next if $mode eq "NEGATIVE" && $bytes > 0;

    my @frames = split(/;/, $stack);
    my @selected;

    for my $frame (@frames) {
        $frame =~ s/<[^<>]*>$//;

        if ($frame =~ /flb_|in_tail|tail_fs_event|cb_|send_loki|cio_|msgpack|CRYPTO_|SSL_|cmt_|unpack_|encoder_|rallocx|sds_|_create/) {
            push @selected, $frame;
        }
    }

    my $path = @selected
        ? join(" > ", @selected)
        : $stack;

    push @rows, [$bytes, substr($path, 0, 650)];
}

close $fh;

die "Unrecognized collapsed profile format\n"
    if $nonempty > 0 && $parsed == 0;

@rows = sort {
    $mode eq "POSITIVE"
        ? $b->[0] <=> $a->[0]
        : $a->[0] <=> $b->[0]
} @rows;

if (!@rows) {
    print "NO_MATCHING_STACKS ($mode)\n";
    exit 0;
}

for (my $i = 0; $i < @rows && $i < $limit; $i++) {
    printf "%.0f\t%s\n",
        $rows[$i]->[0],
        $rows[$i]->[1];
}
PERL_V7

# Validate with the actual Perl runtime inside Analysis Pod.
if ! perl -c /work/stack-top.pl >/work/perl-check.log 2>&1; then
    cat /work/perl-check.log
    exit 43
fi

printf '%s\n' \
  'a;flb_tail_file_append<0x1> 1048576' \
  'a;cb_loki_flush<0x2>;CRYPTO_malloc -524288' \
  > /work/stack-test.txt

positive="$(perl /work/stack-top.pl /work/stack-test.txt POSITIVE 2)" || exit 44
negative="$(perl /work/stack-top.pl /work/stack-test.txt NEGATIVE 2)" || exit 45

expected_positive="$(printf '1048576\tflb_tail_file_append')"
expected_negative="$(printf '%s\t%s' -524288 'cb_loki_flush > CRYPTO_malloc')"

if [ "$positive" != "$expected_positive" ] ||
   [ "$negative" != "$expected_negative" ]; then
    echo "ERROR: V7 formatter self-test failed"
    exit 46
fi

echo "FULL STACK FORMATTER V7 SELFTEST PASSED"

'''

replacement = r'''show_delta() {
  label="$1"
  base="$2"
  current="$3"

  echo
  echo "=== HEAP GROWTH: $label ==="

  if ! timeout 180 jeprof \
      --collapsed \
      --show_bytes \
      --inuse_space \
      --exclude="$FILTER" \
      --base="$base" \
      "$BIN" "$current" \
      > /work/delta.collapsed \
      2> /work/jeprof.err; then

    cat /work/jeprof.err
    exit 23
  fi

  awk '$NF ~ /^-?[0-9]+$/ {s+=$NF}
       END {
         printf "Sampled in-use net difference: %+.2f MiB\n",
                s/1048576
       }' /work/delta.collapsed

  echo "--- TOP POSITIVE FULL PATHS ---"
  perl /work/stack-top.pl /work/delta.collapsed POSITIVE 12

  echo "--- TOP NEGATIVE FULL PATHS ---"
  perl /work/stack-top.pl /work/delta.collapsed NEGATIVE 8
}

'''

s = s.replace(old, new, 1)
s = s.replace(marker, marker + "\n" + setup, 1)

a = s.index(begin)
b = s.index(end, a)
s = s[:a] + replacement + s[b:]

tmp = p.with_name(p.name + ".v7.tmp")
tmp.write_text(s)
shutil.copymode(str(p), str(tmp))

check = subprocess.run(
    ["bash", "-n", str(tmp)],
    stdout=subprocess.PIPE,
    stderr=subprocess.PIPE,
    universal_newlines=True
)

if check.returncode != 0:
    tmp.unlink()
    sys.exit(check.stderr)

os.replace(str(tmp), str(p))

print("V7 PATCH APPLIED")
print("BASH SYNTAX PASSED")
PY

python3 /tmp/flb-jeprof-v7-patch.py
