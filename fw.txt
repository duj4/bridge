jq -r '.data.requested_rules.list[].flowdbIds' "$FILE"

jq '
  [
    .data.requested_rules.list[].flowdbIds
    | select(type == "string")
    | (split("\n") | length - 1)
  ] | add // 0
' "$FILE"

jq '
  [
    .data.requested_rules.list[]
    | select((.flowdbIds // "") | contains(","))
    | {
        id,
        flowdbIds
      }
  ][:10]
' "$FILE"

jq '
  .data.requested_rules.list
  | group_by(.status)
  | map({
      status: .[0].status,
      total: length,
      withFlow: (
        [.[] | select((.flowdbIds // "") != "")]
        | length
      ),
      withoutFlow: (
        [.[] | select((.flowdbIds // "") == "")]
        | length
      )
    })
' "$FILE"
