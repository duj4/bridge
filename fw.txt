jq '
  def has_value($key):
    ((.[$key] // "")
    | tostring
    | gsub("^\\s+|\\s+$"; "")
    | length) > 0;

  (.data.requested_rules.list // []) as $requested
  | (.data.review_rules.list // []) as $review
  | ($requested + $review) as $raw

  # 按 Rule ID 去重，重复时优先使用 requested_rules
  | (
      reduce $raw[] as $r (
        {};
        .[($r.id | tostring)] //= $r
      )
      | [.[]]
    ) as $rules

  # 统计四种互斥情况
  | (
      reduce $rules[] as $r (
        {
          onlyFlowdb: 0,
          onlyLegacy: 0,
          both: 0,
          neither: 0
        };

        if (
          ($r | has_value("flowdbIds")) and
          ($r | has_value("legacyExcelFlowdbIds"))
        ) then
          .both += 1

        elif ($r | has_value("flowdbIds")) then
          .onlyFlowdb += 1

        elif ($r | has_value("legacyExcelFlowdbIds")) then
          .onlyLegacy += 1

        else
          .neither += 1
        end
      )
    ) as $counts

  | ($counts | [
      .onlyFlowdb,
      .onlyLegacy,
      .both,
      .neither
    ] | add) as $sum

  | {
      requestedRules: ($requested | length),
      reviewRules: ($review | length),
      rawCombined: ($raw | length),
      duplicateRecords: (($raw | length) - ($rules | length)),
      uniqueRules: ($rules | length),

      onlyFlowdb: $counts.onlyFlowdb,
      onlyLegacy: $counts.onlyLegacy,
      both: $counts.both,
      neither: $counts.neither,

      classifiedTotal: $sum,
      countMatched: ($sum == ($rules | length)),
      matchesRequestedTotalCount:
        (($rules | length) == .data.requested_rules.totalCount)
    }
' ./rules.json


jq -r '
  def empty_flow($v):
    (($v // "") | tostring
    | gsub("^\\s+|\\s+$"; "")
    | length) == 0;

  .data.requested_rules.list
  | map(
      select(
        empty_flow(.flowdbIds)
        and empty_flow(.legacyExcelFlowdbIds)
      )
    )
  | group_by(.status)[]
  | [.[0].status, length]
  | @tsv
' "$FILE"

jq '
  def empty_flow($v):
    (($v // "") | tostring
    | gsub("^\\s+|\\s+$"; "")
    | length) == 0;

  limit(1;
    .data.requested_rules.list[]
    | select(
        .status == "Active"
        and empty_flow(.flowdbIds)
        and empty_flow(.legacyExcelFlowdbIds)
      )
  )
  | {
      id,
      status,
      type,
      flowdbIds,
      legacyExcelFlowdbIds,
      services,
      ticketId,
      businessJustifications
    }
' "$FILE"
