jq '
  .data.requested_rules.list
  | map(
      select((.flowdbIds // "") | contains(","))
    )
  | {
      total: length,
      byStatus: (
        group_by(.status)
        | map({
            status: .[0].status,
            count: length
          })
      ),
      byType: (
        group_by(.type)
        | map({
            type: .[0].type,
            count: length
          })
      ),
      withLegacyFlowReference: (
        [
          .[]
          | select(
              (.legacyExcelFlowdbIds // "") != ""
            )
        ] | length
      )
    }
' "$FILE"
