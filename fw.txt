jq '
  def flow_id:
    (.flowdbIds // "")
    | tostring
    | gsub("^\\s+|\\s+$"; "");

  .data.requested_rules.list as $rules
  | [$rules[] | flow_id | select(length > 0)] as $refs
  | ($refs | group_by(.)) as $groups
  | {
      totalRules: ($rules | length),
      rulesWithFlow: ($refs | length),
      rulesWithoutFlow: (($rules | length) - ($refs | length)),
      uniqueFlows: ($groups | length),
      flowsWithMultipleRules:
        ([$groups[] | select(length > 1)] | length),
      maxRulesPerFlow:
        ([$groups[] | length] | max // 0),
      suspiciousMultiFlowValues:
        ([$refs[] | select(contains(","))] | length)
    }
' "$FILE"

jq '
  def split_groups:
    (. // "")
    | split(",")
    | map(gsub("^\\s+|\\s+$"; ""))
    | map(select(length > 0));

  .data.requested_rules.list as $rules
  | ([$rules[] | .sourceObjectDetails | split_groups[]]) as $src
  | ([$rules[] | .destinationObjectDetails | split_groups[]]) as $dst
  | {
      uniqueSourceGroups: ($src | unique | length),
      uniqueDestinationGroups: ($dst | unique | length),
      uniqueTotalGroups: (($src + $dst) | unique | length),
      totalGroupReferences: (($src | length) + ($dst | length))
    }
' "$FILE"

jq '
  [.data.requested_rules.list[].id] as $ids
  | {
      reviewCount: (.data.review_rules.list | length),
      reviewAlsoInRequested:
        ([
          .data.review_rules.list[]
          | .id
          | select(. as $id | $ids | index($id) != null)
        ] | length),
      reviewStatuses:
        ([.data.review_rules.list[].status] | unique)
    }
' "$FILE"
